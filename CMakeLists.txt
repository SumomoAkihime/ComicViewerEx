cmake_minimum_required(VERSION 3.24)
project(ComicViewerEx VERSION 1.1.0 LANGUAGES C CXX RC)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
foreach(feature ANIM_UTILS CWEBP DWEBP GIF2WEBP IMG2WEBP VWEBP WEBPINFO LIBWEBPMUX WEBPMUX EXTRAS FUZZTEST)
    set(WEBP_BUILD_${feature} OFF CACHE BOOL "" FORCE)
endforeach()
set(WEBP_LINK_STATIC ON CACHE BOOL "" FORCE)
add_subdirectory(third_party/libwebp EXCLUDE_FROM_ALL)
add_library(sqlite STATIC third_party/sqlite/sqlite3.c)
target_include_directories(sqlite PUBLIC third_party/sqlite)
target_compile_definitions(sqlite PRIVATE SQLITE_THREADSAFE=1 SQLITE_OMIT_LOAD_EXTENSION)
add_library(comic_core STATIC src/common.cpp src/archive.cpp src/image.cpp src/store.cpp src/reader.cpp)
target_include_directories(comic_core PUBLIC src PRIVATE third_party/7zip/CPP third_party/libwebp/src)
target_compile_definitions(comic_core PUBLIC UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
target_compile_options(comic_core PRIVATE /utf-8 /W4 /permissive- /EHsc)
target_link_libraries(comic_core PUBLIC sqlite webp webpdemux ole32 oleaut32 windowscodecs shlwapi propsys)
add_executable(ComicViewerEx WIN32 src/main.cpp src/sidebar.cpp src/tag_bar.cpp src/app.rc)
target_compile_options(ComicViewerEx PRIVATE /utf-8 /W4 /permissive- /EHsc)
target_link_libraries(ComicViewerEx PRIVATE comic_core comctl32 d2d1 dwrite shell32)
target_link_options(ComicViewerEx PRIVATE /MANIFEST:NO)
add_custom_command(TARGET ComicViewerEx POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_SOURCE_DIR}/third_party/7zip-bin/7z.dll" "$<TARGET_FILE_DIR:ComicViewerEx>/7z.dll")
include(CTest)
if(BUILD_TESTING)
    foreach(test store image archive reader)
        if(EXISTS "${CMAKE_SOURCE_DIR}/tests/${test}_test.cpp")
            add_executable(${test}_test tests/${test}_test.cpp)
            target_link_libraries(${test}_test PRIVATE comic_core)
            target_compile_options(${test}_test PRIVATE /utf-8 /W4 /EHsc)
            target_include_directories(${test}_test PRIVATE third_party/libwebp/src)
            add_test(NAME ${test} COMMAND ${test}_test)
        endif()
    endforeach()
endif()
